##GARCH

from arch import arch_model

garch_model = arch_model(
    returns.dropna() * 100,
    mean="Constant",
    vol="GARCH",
    p=1,
    q=1,
    dist="normal"
)

garch_results = garch_model.fit()

print(garch_results.summary())

#Check whether our GARCH model is adequate
# Standardized residuals
standardized_residuals = garch_results.std_resid.dropna()

# ACF of standardized residuals
from statsmodels.graphics.tsaplots import plot_acf

plot_acf(standardized_residuals, lags=30)

plt.title("GARCH Standardized Residuals - ACF")
plt.show()

#squared standardized residuals:

squared_standardized = standardized_residuals **2

plot_acf(squared_standardized, lags = 30)

plt.title("Squared GARCH Standardized Residuals - ACF")
plt.show()

#Ljung-Box test

from statsmodels.stats.diagnostic import acorr_ljungbox

print("Standardized residuals:")
print(acorr_ljungbox(
    standardized_residuals,
    lags=[10,20],
    return_df=True
))

print("\nSquared standardized residuals")
print(acorr_ljungbox(
    squared_standardized, 
    lags=[10,20],
    return_df=True
))
